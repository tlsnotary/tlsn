1//! Future utilities.

3use std::{future::Future, pin::Pin, task::Poll};
4
5/// The state of the auxiliary future after [`poll_with`] returns.
6pub(crate) enum Aux<A: Future> {
7    /// The auxiliary future completed while the main future was polled.
8    Completed(A::Output),
9    /// The auxiliary future is still pending and is returned for later
10    /// examination.
11    Pending(Pin<Box<A>>),
12}
13
14/// Polls `main` to completion while also polling `aux`.
15///
16/// `main` has priority: it is polled first, and if both futures become ready
17/// during the same poll, only `main` completes.
18///
19/// If `aux` completes while `main` is pending, its output is retained and
20/// returned. If `main` completes first, `aux` is not dropped; it is returned
21/// as [`Aux::Pending`] so the caller can examine or continue polling it.
22pub(crate) async fn poll_with<M, A>(main: M, aux: A) -> (M::Output, Aux<A>)
23where
24    M: Future,
25    A: Future,
26{
27    let mut main = Box::pin(main);
28    let mut aux = Box::pin(aux);
29    let mut aux_output = None;
30
31    let main_output = futures::future::poll_fn(|cx| {
32        // `main` is polled first so that it wins if both are ready.
33        if let Poll::Ready(output) = main.as_mut().poll(cx) {
34            return Poll::Ready(output);
35        }
36
37        // Once the auxiliary future completes, stop polling it.
38        if aux_output.is_none() {
39            match aux.as_mut().poll(cx) {
40                Poll::Ready(output) => {
41                    aux_output = Some(output);
42                    // Re-poll `main` in case the auxiliary future's completion
43                    // unblocked it.
44                    cx.waker().wake_by_ref();
45                }
46                Poll::Pending => {}
47            }
48        }
49
50        Poll::Pending
51    })
52    .await;
53
54    let aux = match aux_output {
55        Some(output) => Aux::Completed(output),
56        None => Aux::Pending(aux),
57    };
58
59    (main_output, aux)
60}

62#[cfg(test)]
63mod tests {
64    use std::{
65        cell::Cell,
66        future::{poll_fn, ready},
67        rc::Rc,
68        task::Poll,
69    };
70
71    use super::{Aux, poll_with};
72
73    #[test]
74    fn main_wins_when_both_are_ready() {
75        let aux_polls = Rc::new(Cell::new(0));
76        let aux = {
77            let polls = aux_polls.clone();
78            poll_fn(move |_| {
79                polls.set(polls.get() + 1);
80                Poll::Ready("aux")
81            })
82        };
83
84        let (main, aux) = futures::executor::block_on(poll_with(ready("main"), aux));
85
86        assert_eq!(main, "main");
87        assert_eq!(aux_polls.get(), 0, "aux must not run when main is ready");
88
89        // The auxiliary future is still available for later examination.
90        let Aux::Pending(aux) = aux else {
91            panic!("aux should still be pending");
92        };
93        assert_eq!(futures::executor::block_on(aux), "aux");
94    }
95
96    #[test]
97    fn aux_is_dragged_until_main_completes() {
98        let main_polls = Rc::new(Cell::new(0));
99        let main = {
100            let polls = main_polls.clone();
101            poll_fn(move |_| {
102                polls.set(polls.get() + 1);
103                if polls.get() == 1 {
104                    Poll::Pending
105                } else {
106                    Poll::Ready("main")
107                }
108            })
109        };
110
111        let (main, aux) = futures::executor::block_on(poll_with(main, ready("aux")));
112
113        assert_eq!(main, "main");
114        assert_eq!(main_polls.get(), 2);
115        assert!(
116            matches!(aux, Aux::Completed("aux")),
117            "aux should have completed while main was pending"
118        );
119    }
120
121    #[test]
122    fn main_completion_returns_pending_aux() {
123        let (main, aux) = futures::executor::block_on(poll_with(
124            ready("main"),
125            futures::future::pending::<&str>(),
126        ));
127
128        assert_eq!(main, "main");
129        assert!(matches!(aux, Aux::Pending(_)));
130    }
