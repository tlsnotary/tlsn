33const ALLOWED_GROUPS: &[NamedGroup] = &[NamedGroup::secp256r1];
34const ALLOWED_SUITES: &[CipherSuite] = &[
35    CipherSuite::TLS_ECDHE_RSA_WITH_AES_128_GCM_SHA256,
36    CipherSuite::TLS_ECDHE_ECDSA_WITH_AES_128_GCM_SHA256,
37];

39pub(crate) struct ProxyTlsClient {
40    /// TLS client connection state machine.
41    conn: ClientConnection,
42    /// Master secret captured during the handshake for finalization.
43    ms_log: Arc<MasterSecretLog>,
44    /// Connection timestamp, set when the handshake completes.
45    time: Option<u64>,
46    /// Captured TLS and application traffic for the transcript.
47    traffic: TlsBytes,
48    /// Whether the server transport closed abruptly (EOF/BrokenPipe). This is
49    /// a transport fact rather than close state: it decides whether sending
50    /// our close_notify can still succeed.
51    transport_closed: bool,
52    /// Decrypted application bytes still buffered by rustls.
53    plaintext_bytes_to_read: usize,
54    /// Per-direction close state. Drives the finalize decision.
55    close: CloseState,
56    /// Protocol state machine.
57    state: State,
58    /// Shared decryption toggle.
59    decrypt: Arc<DecryptState>,
60}
61
62enum State {
63    Init {
64        prover: Box<ProxyProver>,
65    },
66    Handshaking {
67        prover: Box<ProxyProver>,
68    },
69    Connected {
70        sent_close_notify: bool,
71        prover: Box<ProxyProver>,
72    },
73    Finalizing {
74        fut: Pin<FinalizeFuture>,
75    },
76    Error,
77}
78
79type FinalizeFuture =
80    Box<dyn Future<Output = Result<(Context, ProverZk, TlsOutput), TlsnError>> + Send>;

82impl ProxyTlsClient {
83    pub(crate) fn new(
84        prover: Box<ProxyProver>,
85        config: &TlsClientConfig,
86        server_name: ServerName,
87    ) -> Result<Self, TlsnError> {
88        let provider = rustls::crypto::ring::default_provider();
89
90        let kx_groups = provider
91            .kx_groups
92            .iter()
93            .filter(|g| ALLOWED_GROUPS.contains(&g.name()))
94            .copied()
95            .collect();
96        let cipher_suites: Vec<SupportedCipherSuite> = provider
97            .cipher_suites
98            .iter()
99            .filter(|s| match s {
100                SupportedCipherSuite::Tls12(tls12) => ALLOWED_SUITES.contains(&tls12.common.suite),
101                _ => false,
102            })
103            .copied()
104            .collect();
105        let provider = CryptoProvider {
106            kx_groups,
107            cipher_suites,
108            ..provider
109        };
110
111        let ms_log = Arc::new(MasterSecretLog::default());
112        let mut config = create_client_config(config, provider)?;
113        config.key_log = ms_log.clone();
114
115        let conn = ClientConnection::new(Arc::new(config), server_name.into_pki_server_name())
116            .map_err(|err| {
117                TlsnError::internal()
118                    .with_msg("rustls error")
119                    .with_source(err)
120            })?;
121
122        let decrypt = DecryptState {
123            decrypt: AtomicBool::new(true),
124        };
125
126        let tls_client = Self {
127            conn,
128            ms_log,
129            time: None,
130            traffic: TlsBytes::default(),
131            transport_closed: false,
132            plaintext_bytes_to_read: 0,
133            close: CloseState::new(),
134            state: State::Init { prover },
135            decrypt: Arc::new(decrypt),
136        };
137
138        Ok(tls_client)
139    }
140}

142impl TlsClient for ProxyTlsClient {
143    type Error = TlsnError;
144
145    fn wants_read_tls(&self) -> bool {
146        self.conn.wants_read()
147    }
148
149    fn wants_write_tls(&self) -> bool {
150        self.conn.wants_write()
151    }
152
153    fn read_tls(&mut self, buf: &[u8]) -> Result<usize, Self::Error> {
154        let mut reader = buf;
155        let n = self
156            .conn
157            .read_tls(&mut reader)
158            .map_err(|e| TlsnError::internal().with_source(e))?;
159        self.traffic.tls_recv.extend_from_slice(&buf[..n]);
160        Ok(n)
161    }
162
163    fn write_tls(&mut self, buf: &mut [u8]) -> Result<usize, Self::Error> {
164        let mut writer = buf as &mut [u8];
165        let n = self
166            .conn
167            .write_tls(&mut writer)
168            .map_err(|e| TlsnError::internal().with_source(e))?;
169        self.traffic.tls_sent.extend_from_slice(&buf[..n]);
170        Ok(n)
171    }
172
173    fn wants_read(&self) -> bool {
174        matches!(self.state, State::Connected { .. })
175    }
176
177    fn wants_write(&self) -> bool {
178        matches!(
179            self.state,
180            State::Handshaking { .. } | State::Connected { .. }
181        ) && !self.close.close_requested()
182    }
183
184    fn read(&mut self, buf: &mut [u8]) -> Result<usize, Self::Error> {
185        let n = match self.conn.reader().read(buf) {
186            Ok(n) => Ok::<_, TlsnError>(n),
187            Err(e) if e.kind() == std::io::ErrorKind::WouldBlock => Ok(0),
188            Err(e) => return Err(TlsnError::internal().with_source(e)),
189        }?;
190        self.traffic.app_recv.extend_from_slice(&buf[..n]);
191        self.plaintext_bytes_to_read = self.plaintext_bytes_to_read.saturating_sub(n);
192        if n == 0 && self.close.peer_closed() {
193            self.close.mark_drained();
194        }
195
196        Ok(n)
197    }
198
199    fn write(&mut self, buf: &[u8]) -> Result<usize, Self::Error> {
200        let reader = buf;
201        let n = self
202            .conn
203            .writer()
204            .write(reader)
205            .map_err(|e| TlsnError::internal().with_source(e))?;
206        self.traffic.app_sent.extend_from_slice(&buf[..n]);
207        Ok(n)
208    }
209
210    fn client_close(&mut self) {
211        self.close.mark_close_requested();
212    }
213
214    fn is_handshaking(&self) -> bool {
215        matches!(self.state, State::Init { .. } | State::Handshaking { .. })
216    }
217
218    fn incoming_finished(&self) -> bool {
219        self.close.peer_closed()
220    }
221
222    fn plaintext_pending(&self) -> bool {
223        self.close.peer_closed() && !self.close.inbound_drained()
224    }
225
226    fn has_pending_plaintext(&self) -> bool {
227        self.plaintext_bytes_to_read > 0 || self.plaintext_pending()
228    }
229
230    fn server_close(&mut self) {
231        self.transport_closed = true;
232        self.close.mark_peer_closed();
233        // Nothing can be written to a dead transport, so the outbound side is
234        // finished from the application's point of view.
235        self.close.mark_close_requested();
236    }
237
238    fn wants_write_shutdown(&self) -> bool {
239        matches!(self.state, State::Connected { .. })
240            && self.close.close_requested()
241            && !self.close.outbound_flushed()
242            && !self.conn.wants_write()
243    }
244
245    fn write_shutdown_complete(&mut self) {
246        self.close.mark_outbound_flushed();
247        self.close.mark_outbound_closed();
248    }
249
250    fn decrypt(&self) -> Arc<DecryptState> {
251        self.decrypt.clone()
252    }
253
254    fn poll(
255        &mut self,
256        cx: &mut std::task::Context,
257    ) -> std::task::Poll<Result<(Context, ProverZk, TlsOutput), Self::Error>> {
258        match std::mem::replace(&mut self.state, State::Error) {
259            State::Init { prover } => {
260                debug!("inner client is starting");
261                self.state = State::Handshaking { prover };
262                self.poll(cx)
263            }
264            State::Handshaking { prover } => {
265                debug!("inner client is handshaking");
266                let io_state = self.conn.process_new_packets().map_err(|err| {
267                    TlsnError::internal()
268                        .with_msg("rustls error")
269                        .with_source(err)
270                })?;
271                self.plaintext_bytes_to_read = io_state.plaintext_bytes_to_read();
272                if !self.conn.is_handshaking() {
273                    let now = web_time::UNIX_EPOCH
274                        .elapsed()
275                        .expect("system time is available")
276                        .as_secs();
277                    self.time = Some(now);
278
279                    debug!("inner client is transitioning to connected");
280                    self.state = State::Connected {
281                        sent_close_notify: false,
282                        prover,
283                    };
284                    self.poll(cx)
285                } else if self.transport_closed {
286                    Poll::Ready(Err(
287                        TlsnError::io().with_msg("server closed during TLS handshake")
288                    ))
289                } else if self.close.close_requested() {
290                    // A client that abandons the handshake leaves no data to
291                    // drive rustls and no socket to wake on, so the state
292                    // machine would stay pending forever.
293                    Poll::Ready(Err(
294                        TlsnError::io().with_msg("client closed during TLS handshake")
295                    ))
296                } else {
297                    self.state = State::Handshaking { prover };
298                    Poll::Pending
299                }
300            }
301            State::Connected {
302                mut sent_close_notify,
303                prover,
304            } => {
305                trace!("inner client is connected");
306                let io_state = self.conn.process_new_packets().map_err(|err| {
307                    TlsnError::internal()
308                        .with_msg("rustls error")
309                        .with_source(err)
310                })?;
311                self.plaintext_bytes_to_read = io_state.plaintext_bytes_to_read();
312
313                if io_state.peer_has_closed() && !self.close.peer_closed() {
314                    self.close.mark_peer_closed();
315                    // TLS 1.2 requires an immediate close_notify response and
316                    // that pending writes be discarded. Writes already handed
317                    // to rustls may still be in flight; from here on we stop
318                    // accepting new application plaintext.
319                    self.close.mark_close_requested();
320                    debug!("server closed its TLS write side; responding with close_notify");
321                }
322
323                // `wants_read` is false once the state leaves `Connected`, so
324                // only mark the inbound side drained while it is still safe to
325                // deliver the remaining plaintext.
326                if self.close.peer_closed() && io_state.plaintext_bytes_to_read() == 0 {
327                    self.close.mark_drained();
328                }
329
330                if self.close.close_requested() && !self.transport_closed && !sent_close_notify {
331                    debug!("sent close notify");
332                    self.conn.send_close_notify();
333                    sent_close_notify = true;
334                }
335
336                // Send ProxyFinished once, when both directions are finished:
337                // the prover's own close_notify has been handed to the proxy
338                // stream and the write half is closed, and the server has
339                // either sent its TLS close_notify or the proxy stream reached
340                // EOF. The counts are final at this point.
341                //
342                // Do not finalize while rustls still holds decrypted plaintext:
343                // `wants_read` is false once the state leaves `Connected`, so
344                // the prover would never hand that plaintext to the
345                // application and the response would be lost.
346                let finalize_now = self.close.can_finalize();
347
348                if finalize_now {
349                    let ms = self.ms_log.take();
350                    if ms.is_empty() {
351                        return Poll::Ready(Err(
352                            TlsnError::internal().with_msg("master secret is not available")
353                        ));
354                    }
355                    let time = self.time.ok_or_else(|| {
356                        TlsnError::internal().with_msg("connection timestamp is not set")
357                    })?;
358                    let traffic = std::mem::take(&mut self.traffic);
359                    let tls_sent = traffic.tls_sent.len() as u64;
360                    let tls_recv = traffic.tls_recv.len() as u64;
361
362                    let fut = Box::pin(async move {
363                        let mut prover = prover;
364                        prover.send_finished(tls_sent, tls_recv).await?;
365                        prover.finalize(ms, time, traffic).await
366                    });
367
368                    self.state = State::Finalizing { fut };
369                    self.poll(cx)
370                } else {
371                    self.state = State::Connected {
372                        sent_close_notify,
373                        prover,
374                    };
375                    Poll::Pending
376                }
377            }
378            State::Finalizing { mut fut } => {
379                debug!("inner client is finalizing");
380                match fut.poll_unpin(cx)? {
381                    Poll::Ready(output) => Poll::Ready(Ok(output)),
382                    Poll::Pending => {
383                        self.state = State::Finalizing { fut };
384                        Poll::Pending
385                    }
386                }
387            }
388            State::Error => Poll::Ready(Err(
389                TlsnError::internal().with_msg("tls proxy client is in error state")
390            )),
391        }
392    }
393}

395fn create_client_config(
396    config: &TlsClientConfig,
397    provider: CryptoProvider,
398) -> Result<rustls::ClientConfig, TlsnError> {
399    let mut root_store = RootCertStore::empty();
400    for cert in &config.root_store().roots {
401        let der = CertificateDer::from_slice(&cert.0);
402        let anchor = anchor_from_trusted_cert(&der)
403            .map_err(|e| {
404                TlsnError::config()
405                    .with_msg("failed to parse root certificate")
406                    .with_source(e)
407            })?
408            .to_owned();
409        root_store.roots.push(anchor);
410    }
411
412    let builder = rustls::ClientConfig::builder_with_provider(Arc::new(provider))
413        .with_protocol_versions(&[&rustls::version::TLS12])
414        .map_err(|e| {
415            TlsnError::config()
416                .with_msg("failed to set protocol versions")
417                .with_source(e)
418        })?
419        .with_root_certificates(root_store);
420
421    let mut rustls_config = if let Some((cert, key)) = config.client_auth() {
422        builder
423            .with_client_auth_cert(
424                cert.iter()
425                    .map(|cert| CertificateDer::from(cert.0.clone()))
426                    .collect(),
427                PrivateKeyDer::try_from(key.0.clone()).map_err(|e| {
428                    TlsnError::config()
429                        .with_msg("failed to parse private key")
430                        .with_source(e)
431                })?,
432            )
433            .map_err(|e| {
434                TlsnError::config()
435                    .with_msg("failed to configure client authentication")
436                    .with_source(e)
437            })?
438    } else {
439        builder.with_no_client_auth()
440    };
441
442    // disable session resumption, due to more complicated security assumptions
443    rustls_config.resumption = Resumption::disabled();
444
445    Ok(rustls_config)
446}
