32pin_project_lite::pin_project! {
33    /// State during the MPC-TLS connection.
34    ///
35    /// TLS bytes reach the peer in two hops: the client fills a staging
36    /// channel and the prover forwards from there to the transport. Records the
37    /// client produced can therefore be in flight between the two, so the
38    /// transport must not be closed until that channel is both closed and
39    /// drained.
40    ///
41    /// The field comments below are plain comments because this struct is
42    /// expanded by `pin_project_lite`, which rejects attributes other than
43    /// `#[pin]` on fields; see taiki-e/pin-project-lite#3.
44    #[project = ConnectedProj]
45    pub struct Connected<S> {
46        pub(crate) server_name: ServerName,
47        pub(crate) tls_client: Box<dyn TlsClient<Error = Error> + Send>,
48        pub(crate) control: ProverControl,
49        #[pin]
50        pub(crate) client_io: DuplexStream,
51        pub(crate) output: Option<(Context, ProverZk, TlsOutput)>,
52        // Transport carrying TLS bytes towards the peer. In proxy mode this is
53        // the mux stream to the verifier, which relays it to the server.
54        #[pin]
55        pub(crate) server_socket: S,
56        // The client's end of the staging channel. The TLS client writes its
57        // records here; they surface on `socket_end` for forwarding to
58        // `server_socket`. Bytes arriving from the server are written to
59        // `socket_end` and read from here.
60        #[pin]
61        pub(crate) client_end: DuplexStream,
62        // The socket's end of the same channel: outbound bytes written to
63        // `client_end` surface here and are forwarded to `server_socket`, and
64        // inbound bytes read from `server_socket` are staged here for the
65        // client to read.
66        #[pin]
67        pub(crate) socket_end: DuplexStream,
68        // Per-direction close state at the transport layer.
69        pub(crate) close: CloseState,
70        // Whether the server transport itself reached EOF/reset. Kept separate
71        // from the inbound close state: delivering app EOF does not mean the
72        // transport is gone.
73        pub(crate) server_transport_closed: bool,
74        // Whether `client_end` has been closed for writing. That marks the end
75        // of the TLS output stream: the channel only reports end-of-file once
76        // every byte has been read out, which is what makes it a reliable
77        // "everything has been handed to the transport" signal.
78        pub(crate) pipe_closed: bool,
79        // Whether those bytes have all been forwarded to `server_socket`.
80        pub(crate) pipe_drained: bool
81    }
82}
